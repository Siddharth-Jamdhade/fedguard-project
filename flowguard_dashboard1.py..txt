"""
FlowGuard+ — Complete Live Dashboard

ESP32-C3 + HC-SR04
        ↓
distance_cm
        ↓
water_level_cm = 13.53 - distance_cm
        ↓
Physics blockage detection
        ↓
ML confirmation
        ↓
Live dashboard

Also retains:
- Cd calibration
- Manual testing
- Network cascade simulation
- Muskingum routing
- Audit trail / database
- Blockage history
- Network simulation history
"""

import time
from collections import deque

import pandas as pd
import streamlit as st

from blockage_detector import (
    calibrate_cd,
    calculate_area,
    blockage_percent,
    extract_window_features,
    BlockageAnomalyDetector,
    forecast_days_to_critical,
    ESP32SensorReader,
    parse_sensor_csv_line,
)

from network_simulation import (
    WaterNetwork,
    generate_rainfall_pulse,
)

from storage import (
    init_db,
    log_calibration,
    log_reading,
    log_blockage_event,
    log_network_run,
    get_calibration_log,
    get_readings,
    get_blockage_events,
    get_network_runs,
)


# ============================================================
# PAGE
# ============================================================

st.set_page_config(
    page_title="FlowGuard+ Live Monitor",
    page_icon="🌊",
    layout="wide",
)


# ============================================================
# CONSTANTS
# ============================================================

# YOUR REAL SENSOR HEIGHT
SENSOR_MOUNT_HEIGHT_CM = 13.53

# YOUR KNOWN CLEAN PIPE/CHANNEL AREA
CLEAN_AREA_CM2 = 1.90

# Default serial settings
DEFAULT_BAUDRATE = 115200


# ============================================================
# DATABASE
# ============================================================

init_db()


# ============================================================
# SESSION STATE
# ============================================================

defaults = {
    "cd": None,
    "a_clean": CLEAN_AREA_CM2,
    "sensor_reader": None,
    "sensor_connected": False,
    "sensor_port": "COM5",
    "last_reading": None,
    "live_history": deque(maxlen=120),
    "network_nodes": None,
    "rainfall": None,
}

for key, value in defaults.items():
    if key not in st.session_state:
        st.session_state[key] = value


# ============================================================
# HEADER
# ============================================================

st.title("🌊 FlowGuard+ — Cascading Flood Early Warning")

st.caption(
    "ESP32-C3 + HC-SR04 → Physics → ML → Network Simulation"
)

st.divider()


# ============================================================
# SIDEBAR
# ============================================================

st.sidebar.header("⚙️ FlowGuard Configuration")


# ------------------------------------------------------------
# SENSOR INFORMATION
# ------------------------------------------------------------

st.sidebar.subheader("HC-SR04 Sensor")

st.sidebar.info(
    f"""
Sensor empty-channel height: **{SENSOR_MOUNT_HEIGHT_CM:.2f} cm**

Clean pipe/channel area: **{CLEAN_AREA_CM2:.2f} cm²**

Water level:

**13.53 − measured distance**
"""
)


# ------------------------------------------------------------
# SERIAL PORT
# ------------------------------------------------------------

sensor_port = st.sidebar.text_input(
    "ESP32 COM Port",
    value=st.session_state["sensor_port"],
)

st.session_state["sensor_port"] = sensor_port


baudrate = st.sidebar.number_input(
    "Baud rate",
    min_value=1200,
    max_value=921600,
    value=DEFAULT_BAUDRATE,
    step=1200,
)


# ------------------------------------------------------------
# FLOW INPUT
# ------------------------------------------------------------

st.sidebar.subheader("Hydraulic Parameters")

q_cm3_s = st.sidebar.number_input(
    "Inflow Q (cm³/s)",
    min_value=0.001,
    value=50.0,
    step=1.0,
)


# ============================================================
# CALIBRATION
# ============================================================

st.sidebar.subheader("Cd Calibration")

pour_volume = st.sidebar.number_input(
    "Pour volume (mL)",
    min_value=1.0,
    value=500.0,
)

pour_time = st.sidebar.number_input(
    "Pour time (seconds)",
    min_value=0.1,
    value=10.0,
)

steady_h = st.sidebar.number_input(
    "Steady water height (cm)",
    min_value=0.1,
    value=3.0,
)


if st.sidebar.button(
    "Calibrate Cd",
    use_container_width=True,
):

    try:

        cd_value = calibrate_cd(
            pour_volume,
            pour_time,
            steady_h,
            CLEAN_AREA_CM2,
        )

        st.session_state["cd"] = cd_value

        st.session_state["a_clean"] = CLEAN_AREA_CM2

        log_calibration(
            cd_value,
            pour_volume,
            pour_time,
            steady_h,
            CLEAN_AREA_CM2,
        )

        st.sidebar.success(
            f"Cd calibrated = {cd_value:.4f}"
        )

    except Exception as exc:

        st.sidebar.error(
            f"Calibration error: {exc}"
        )


# ------------------------------------------------------------
# CURRENT CD
# ------------------------------------------------------------

cd_active = st.session_state.get("cd")

if cd_active is None:

    st.sidebar.warning(
        "Cd is not calibrated yet."
    )

else:

    st.sidebar.success(
        f"Current Cd = {cd_active:.4f}"
    )


# ============================================================
# ALERT THRESHOLDS
# ============================================================

st.sidebar.subheader("Alert Thresholds")

physics_alert_threshold = st.sidebar.number_input(
    "Physics alert (%)",
    min_value=0.0,
    max_value=100.0,
    value=15.0,
    step=1.0,
)

critical_threshold = st.sidebar.number_input(
    "Critical blockage (%)",
    min_value=1.0,
    max_value=100.0,
    value=50.0,
    step=1.0,
)


# ============================================================
# SECTION 1 — LIVE PHYSICAL NODE
# ============================================================

st.header("📡 Live Physical Node")

st.write(
    "Real HC-SR04 measurements from the ESP32-C3 are used "
    "for the blockage calculation."
)


# ============================================================
# CONNECT / DISCONNECT
# ============================================================

connect_col, disconnect_col, status_col = st.columns(3)


with connect_col:

    connect = st.button(
        "🔌 Connect ESP32",
        use_container_width=True,
    )


with disconnect_col:

    disconnect = st.button(
        "⛔ Disconnect",
        use_container_width=True,
    )


with status_col:

    if st.session_state["sensor_connected"]:

        st.success("🟢 ESP32 CONNECTED")

    else:

        st.warning("⚪ ESP32 DISCONNECTED")


# ------------------------------------------------------------
# CONNECT
# ------------------------------------------------------------

if connect:

    try:

        reader = ESP32SensorReader(
            port=sensor_port,
            baudrate=int(baudrate),
            timeout=1.0,
        )

        reader.connect()

        st.session_state["sensor_reader"] = reader

        st.session_state["sensor_connected"] = True

        st.success(
            f"Connected to {sensor_port}"
        )

    except Exception as exc:

        st.session_state["sensor_connected"] = False

        st.error(
            f"Could not connect to {sensor_port}: {exc}"
        )


# ------------------------------------------------------------
# DISCONNECT
# ------------------------------------------------------------

if disconnect:

    try:

        reader = st.session_state.get(
            "sensor_reader"
        )

        if reader is not None:

            reader.close()

    except Exception:

        pass

    st.session_state["sensor_reader"] = None

    st.session_state["sensor_connected"] = False

    st.info(
        "ESP32 disconnected."
    )


# ============================================================
# MANUAL TEST MODE
# ============================================================

with st.expander(
    "🧪 Manual Reading / Testing Mode"
):

    st.write(
        "This mode is only for testing the physics calculation. "
        "Normal operation should use the HC-SR04."
    )

    manual_height = st.number_input(
        "Manual water level (cm)",
        min_value=0.0,
        value=3.0,
        step=0.1,
    )

    if st.button(
        "Calculate Manual Blockage"
    ):

        if cd_active is None:

            st.error(
                "Calibrate Cd first."
            )

        else:

            area = calculate_area(
                q_cm3_s,
                cd_active,
                manual_height,
            )

            blockage = blockage_percent(
                area,
                CLEAN_AREA_CM2,
            )

            log_reading(
                "Physical_Node_1",
                manual_height,
                q_cm3_s,
                area,
                blockage,
            )

            st.session_state["last_reading"] = {

                "source": "Manual",

                "distance_cm": None,

                "water_level_cm": manual_height,

                "area_cm2": area,

                "blockage_pct": blockage,

                "ml_confirmed": False,

                "forecast": None,
            }


# ============================================================
# LIVE SENSOR READING
# ============================================================

if st.session_state["sensor_connected"]:

    if cd_active is None:

        st.warning(
            "Connect is successful, but you must calibrate Cd "
            "before blockage calculation can run."
        )

    else:

        reader = st.session_state["sensor_reader"]

        try:

            raw_line = reader.serial.readline()

            if raw_line:

                line = raw_line.decode(
                    "utf-8",
                    errors="ignore",
                ).strip()


                # ------------------------------------------------
                # Parse ESP32 CSV
                #
                # Expected:
                #
                # t_ms,distance_cm,water_level_cm
                #
                # Example:
                #
                # 5000,10.42,3.11
                # ------------------------------------------------

                sensor_data = parse_sensor_csv_line(
                    line
                )


                if sensor_data is not None:

                    distance_cm = sensor_data[
                        "distance_cm"
                    ]

                    water_level_cm = sensor_data[
                        "water_level_cm"
                    ]


                    # ------------------------------------------------
                    # SAFETY / VALIDATION
                    # ------------------------------------------------

                    if distance_cm < 0:

                        st.warning(
                            "HC-SR04 returned an invalid distance."
                        )

                    elif water_level_cm < 0:

                        st.warning(
                            "Invalid water level."
                        )

                    else:

                        # ------------------------------------------------
                        # PHYSICS CALCULATION
                        # ------------------------------------------------

                        effective_area = calculate_area(
                            q_cm3_s,
                            cd_active,
                            water_level_cm,
                        )


                        blockage = blockage_percent(
                            effective_area,
                            CLEAN_AREA_CM2,
                        )


                        # ------------------------------------------------
                        # KEEP HISTORY
                        # ------------------------------------------------

                        st.session_state[
                            "live_history"
                        ].append(

                            {
                                "t_ms":
                                    sensor_data["t_ms"],

                                "distance_cm":
                                    distance_cm,

                                "water_level_cm":
                                    water_level_cm,

                                "area_cm2":
                                    effective_area,

                                "blockage_pct":
                                    blockage,
                            }

                        )


                        # ------------------------------------------------
                        # DATABASE
                        # ------------------------------------------------

                        log_reading(
                            "Physical_Node_1",
                            water_level_cm,
                            q_cm3_s,
                            effective_area,
                            blockage,
                        )


                        # ------------------------------------------------
                        # ML CONFIRMATION
                        # ------------------------------------------------

                        ml_confirmed = False

                        forecast = None


                        try:

                            db_readings = get_readings(
                                "Physical_Node_1"
                            )


                            if (

                                not db_readings.empty

                                and
                                "blockage_pct"
                                in db_readings.columns

                            ):

                                blockage_values = (

                                    pd.to_numeric(
                                        db_readings[
                                            "blockage_pct"
                                        ],
                                        errors="coerce",
                                    )
                                    .dropna()
                                    .tolist()

                                )


                                # ----------------------------------------
                                # Use latest 5 readings
                                # ----------------------------------------

                                if len(
                                    blockage_values
                                ) >= 5:

                                    recent = (
                                        blockage_values[-5:]
                                    )


                                    features = (
                                        extract_window_features(
                                            recent
                                        )
                                    )


                                    # ------------------------------------
                                    # ML baseline
                                    # ------------------------------------

                                    baseline = (
                                        blockage_values[:-5]
                                    )


                                    if len(
                                        baseline
                                    ) >= 25:

                                        baseline_windows = []

                                        for i in range(
                                            len(baseline) - 4
                                        ):

                                            window = (
                                                baseline[
                                                    i:i + 5
                                                ]
                                            )

                                            feature = (
                                                extract_window_features(
                                                    window
                                                )
                                            )

                                            if feature is not None:

                                                baseline_windows.append(
                                                    feature
                                                )


                                        if len(
                                            baseline_windows
                                        ) >= 5:

                                            detector = (
                                                BlockageAnomalyDetector(
                                                    contamination=0.1
                                                )
                                            )

                                            detector.fit(
                                                baseline_windows
                                            )


                                            ml_confirmed = (
                                                detector.is_confirmed_anomaly(
                                                    features
                                                )
                                            )


                                    # ------------------------------------
                                    # Forecast
                                    # ------------------------------------

                                    forecast = (
                                        forecast_days_to_critical(
                                            recent,
                                            list(
                                                range(
                                                    len(recent)
                                                )
                                            ),
                                            critical_threshold_pct=(
                                                critical_threshold
                                            ),
                                        )
                                    )

                        except Exception:

                            # ML should never stop the sensor
                            # pipeline if there is not enough data.
                            ml_confirmed = False


                        # ------------------------------------------------
                        # EVENT LOGGING
                        # ------------------------------------------------

                        if (

                            blockage is not None

                            and
                            blockage >= physics_alert_threshold

                        ):

                            log_blockage_event(
                                "Physical_Node_1",
                                blockage,
                                int(ml_confirmed),
                                forecast,
                            )


                        # ------------------------------------------------
                        # SAVE LATEST
                        # ------------------------------------------------

                        st.session_state[
                            "last_reading"
                        ] = {

                            "source":
                                "HC-SR04 / ESP32-C3",

                            "distance_cm":
                                distance_cm,

                            "water_level_cm":
                                water_level_cm,

                            "area_cm2":
                                effective_area,

                            "blockage_pct":
                                blockage,

                            "ml_confirmed":
                                ml_confirmed,

                            "forecast":
                                forecast,

                            "t_ms":
                                sensor_data["t_ms"],
                        }


        except Exception as exc:

            st.error(
                f"Sensor reading error: {exc}"
            )


# ============================================================
# DISPLAY CURRENT READING
# ============================================================

last = st.session_state.get(
    "last_reading"
)


if last is not None:

    blockage = last[
        "blockage_pct"
    ]

    water_level = last[
        "water_level_cm"
    ]

    distance = last[
        "distance_cm"
    ]

    area = last[
        "area_cm2"
    ]

    ml_confirmed = last[
        "ml_confirmed"
    ]


    # ----------------------------------------------------------
    # STATUS
    # ----------------------------------------------------------

    if blockage is None:

        status = "⚪ NO VALID READING"

    elif blockage >= critical_threshold:

        status = "🔴 CRITICAL BLOCKAGE"

    elif blockage >= physics_alert_threshold:

        status = "🟠 BLOCKAGE ALERT"

    else:

        status = "🟢 CHANNEL NORMAL"


    st.subheader(status)


    # ----------------------------------------------------------
    # METRICS
    # ----------------------------------------------------------

    c1, c2, c3, c4 = st.columns(4)


    with c1:

        st.metric(
            "Sensor Distance",
            (
                f"{distance:.2f} cm"
                if distance is not None
                else "N/A"
            ),
        )


    with c2:

        st.metric(
            "Water Level",
            (
                f"{water_level:.2f} cm"
                if water_level is not None
                else "N/A"
            ),
        )


    with c3:

        st.metric(
            "Effective Area",
            (
                f"{area:.3f} cm²"
                if area is not None
                else "N/A"
            ),
        )


    with c4:

        st.metric(
            "Blockage",
            (
                f"{blockage:.1f}%"
                if blockage is not None
                else "N/A"
            ),
        )


    # ----------------------------------------------------------
    # ML
    # ----------------------------------------------------------

    if ml_confirmed:

        st.error(
            "🤖 ML CONFIRMED: sustained abnormal blockage pattern"
        )

    else:

        st.success(
            "🤖 ML: no confirmed anomaly"
        )


    # ----------------------------------------------------------
    # FORECAST
    # ----------------------------------------------------------

    if last["forecast"] is not None:

        st.info(
            f"📈 Forecast: approximately "
            f"{last['forecast']} days to the "
            f"{critical_threshold:.0f}% critical threshold."
        )


    # ----------------------------------------------------------
    # FORMULA DISPLAY
    # ----------------------------------------------------------

    with st.expander(
        "🔬 Current physics calculation"
    ):

        st.write(
            f"""
**Sensor reference height:** {SENSOR_MOUNT_HEIGHT_CM:.2f} cm

**Observed sensor distance:** {distance:.2f} cm

**Water level:**

`h = 13.53 − {distance:.2f}`

`h = {water_level:.2f} cm`

**Known clean area:**

`A_clean = {CLEAN_AREA_CM2:.2f} cm²`

**Effective flow area:**

`A_effective = {area:.3f} cm²`

**Blockage:**

`Blockage % = (1 − A_effective / A_clean) × 100`

`Blockage = {blockage:.2f}%`
"""
        )


# ============================================================
# LIVE GRAPH
# ============================================================

st.subheader(
    "📈 Live Sensor History"
)


history = list(
    st.session_state[
        "live_history"
    ]
)


if len(history) > 1:

    chart_df = pd.DataFrame(
        history
    )

    chart_df = chart_df[
        [
            "water_level_cm",
            "blockage_pct",
        ]
    ]

    chart_df.columns = [
        "Water Level (cm)",
        "Blockage (%)",
    ]

    st.line_chart(
        chart_df
    )

else:

    st.info(
        "Waiting for multiple HC-SR04 readings..."
    )


# ============================================================
# SECTION 2 — NETWORK SIMULATION
# ============================================================

st.divider()

st.header(
    "🌊 Network Cascade Simulation"
)

st.caption(
    "Software simulation: Ambazari Lake → Nag River Segment 1 "
    "→ Nag River Segment 2"
)


network_col1, network_col2 = st.columns(
    [1, 2]
)


with network_col1:

    peak_rain = st.slider(
        "Peak rainfall inflow",
        min_value=20,
        max_value=200,
        value=100,
    )


    if st.button(
        "▶ Run Network Simulation",
        use_container_width=True,
    ):

        network = WaterNetwork()


        network.add_node(
            "Ambazari Lake",
            k=2.0,
            x=0.2,
            node_type="lake",
        )


        network.add_node(
            "Nag River Segment 1",
            k=3.0,
            x=0.25,
            node_type="drain",
        )


        network.add_node(
            "Nag River Segment 2 (downstream)",
            k=4.0,
            x=0.3,
            node_type="drain",
        )


        rainfall = generate_rainfall_pulse(
            duration_steps=30,
            peak_step=10,
            peak_value=peak_rain,
            base_value=5,
        )


        nodes = network.simulate(
            rainfall,
            dt=1.0,
        )


        st.session_state[
            "network_nodes"
        ] = nodes


        st.session_state[
            "rainfall"
        ] = rainfall


        last_node = nodes[-1]


        delay = int(
            last_node.outflow.argmax()
            - rainfall.argmax()
        )


        log_network_run(
            rainfall.max(),
            len(nodes),
            last_node.outflow.max(),
            delay,
        )


with network_col2:

    nodes = st.session_state.get(
        "network_nodes"
    )

    rainfall = st.session_state.get(
        "rainfall"
    )


    if nodes is not None:

        chart_data = pd.DataFrame(
            {
                "Rainfall Inflow":
                    rainfall
            }
        )


        for node in nodes:

            chart_data[
                node.name
            ] = node.outflow


        st.line_chart(
            chart_data
        )


        st.info(
            "Muskingum routing shows how the flood wave "
            "is delayed and smoothed as it moves downstream."
        )


# ============================================================
# SECTION 3 — AUDIT TRAIL
# ============================================================

st.divider()

st.header(
    "🗃️ Audit Trail"
)

st.caption(
    "Stored locally in flowguard_history.db"
)


tab1, tab2, tab3, tab4 = st.tabs(
    [
        "Calibration Log",
        "Blockage Readings",
        "Blockage Events",
        "Network Runs",
    ]
)


with tab1:

    df = get_calibration_log()

    if not df.empty:

        st.dataframe(
            df,
            hide_index=True,
            use_container_width=True,
        )

    else:

        st.info(
            "No calibration records."
        )


with tab2:

    df = get_readings()

    if not df.empty:

        st.dataframe(
            df,
            hide_index=True,
            use_container_width=True,
        )

    else:

        st.info(
            "No readings recorded."
        )


with tab3:

    df = get_blockage_events()

    if not df.empty:

        st.dataframe(
            df,
            hide_index=True,
            use_container_width=True,
        )

    else:

        st.info(
            "No blockage events recorded."
        )


with tab4:

    df = get_network_runs()

    if not df.empty:

        st.dataframe(
            df,
            hide_index=True,
            use_container_width=True,
        )

    else:

        st.info(
            "No network simulations recorded."
        )


# ============================================================
# AUTO REFRESH
# ============================================================

if st.session_state["sensor_connected"]:

    time.sleep(0.1)

    st.rerun()
